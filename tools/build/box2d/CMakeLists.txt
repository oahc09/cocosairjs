cmake_minimum_required(VERSION 3.22)
project(air_box2d LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 20)
set(BOX2D_DISABLE_SIMD ON CACHE BOOL "Portable scalar WASM" FORCE)
set(BOX2D_VALIDATE ON CACHE BOOL "Validate integration" FORCE)
set(BOX2D_SAMPLES OFF CACHE BOOL "Host supplied by AIR" FORCE)
set(BOX2D_UNIT_TESTS OFF CACHE BOOL "Separate upstream target" FORCE)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_compile_options(-ffile-prefix-map=${ROOT}=. -ffp-contract=off)
set(UPSTREAM "${ROOT}/vendor/box2d/source")
add_subdirectory("${UPSTREAM}" box2d)
target_compile_definitions(box2d PRIVATE B2_ENABLE_ASSERT)
target_compile_options(box2d PRIVATE -ffile-prefix-map=${ROOT}=. -ffp-contract=off)
add_executable(air-box2d bridge.cpp contact-material.c)
target_include_directories(air-box2d PRIVATE "${UPSTREAM}/src")
target_link_libraries(air-box2d PRIVATE box2d)
target_compile_options(air-box2d PRIVATE -ffile-prefix-map=${ROOT}=. -ffp-contract=off)
target_link_options(air-box2d PRIVATE --bind -sMODULARIZE=1 -sEXPORT_ES6=1 -sENVIRONMENT=web,node -sALLOW_MEMORY_GROWTH=1 -sASSERTIONS=1 -sFILESYSTEM=0 -sSTACK_SIZE=1048576)
set_target_properties(air-box2d PROPERTIES SUFFIX ".mjs")

# Official scenarios compile unchanged, including their original Sample::Step.
# Only desktop window/file-dialog/draw boundaries are replaced by the browser host.
add_library(air-imgui STATIC ${ROOT}/vendor/box2d/imgui/imgui.cpp ${ROOT}/vendor/box2d/imgui/imgui_draw.cpp ${ROOT}/vendor/box2d/imgui/imgui_tables.cpp ${ROOT}/vendor/box2d/imgui/imgui_widgets.cpp ${ROOT}/vendor/box2d/implot/implot.cpp ${ROOT}/vendor/box2d/implot/implot_items.cpp)
target_include_directories(air-imgui PUBLIC ${ROOT}/vendor/box2d/imgui ${ROOT}/vendor/box2d/implot)
add_subdirectory(${UPSTREAM}/shared shared)
file(GLOB OFFICIAL_SCENARIOS CONFIGURE_DEPENDS ${UPSTREAM}/samples/sample_*.cpp)
list(REMOVE_ITEM OFFICIAL_SCENARIOS ${UPSTREAM}/samples/sample_replay.cpp)
add_executable(air-box2d-samples samples.cpp ${OFFICIAL_SCENARIOS} ${UPSTREAM}/samples/sample.cpp ${UPSTREAM}/samples/car.cpp ${UPSTREAM}/samples/donut.cpp ${UPSTREAM}/samples/doohickey.cpp ${UPSTREAM}/samples/dynamic_mover.cpp ${UPSTREAM}/samples/geometric_mover.cpp)
target_include_directories(air-box2d-samples PRIVATE ${CMAKE_CURRENT_LIST_DIR} ${UPSTREAM}/samples ${UPSTREAM}/extern/jsmn)
target_link_libraries(air-box2d-samples PRIVATE box2d shared air-imgui)
target_compile_options(air-box2d-samples PRIVATE -ffile-prefix-map=${ROOT}=. -ffp-contract=off)
target_compile_definitions(air-box2d-samples PRIVATE B2_ENABLE_ASSERT)
target_link_options(air-box2d-samples PRIVATE --bind -sMODULARIZE=1 -sEXPORT_ES6=1 -sENVIRONMENT=web,node -sALLOW_MEMORY_GROWTH=1 -sASSERTIONS=1 -sSTACK_SIZE=2097152)
set_target_properties(air-box2d-samples PROPERTIES SUFFIX ".mjs")
